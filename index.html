const SS_ID = "1kPKu3SBQ6m8KHRhR-4pvtK6sSVzpKlFi8-A-ru8pfV4"; // mapping tab ikut tingkatan const TAB_BY_TING = { "2": "F22026", "3": "F32026", "4": "F42026" }; // Field yang kita nak return const WANTED_HEADERS = [ "IC", "NAMA", "KELAS_2025", "KELAS_2026", "JANTINA", "GURU_TINGKATAN_2026" ]; function doGet(e) { const ting = String(e?.parameter?.ting || ""); const icRaw = String(e?.parameter?.ic || ""); const mode = String(e?.parameter?.mode || ""); if (mode !== "kelas") { return json_({ ok: false, message: "Mode tak sah. Gunakan mode=kelas" }); } const tabName = TAB_BY_TING[ting]; if (!tabName) { return json_({ ok: false, message: "Tingkatan tidak sah (hanya 2, 3, 4)." }); } const res = findStudentSmart_(tabName, icRaw); return json_(res); } function findStudentSmart_(tabName, icRaw) { const ic = normalizeIC_(icRaw); if (!ic) return { ok: false, message: "IC kosong." }; if (ic.length !== 12) return { ok: false, message: "IC mesti 12 digit." }; const ss = SpreadsheetApp.openById(SS_ID); const sh = ss.getSheetByName(tabName); if (!sh) return { ok: false, message: `Tab ${tabName} tak wujud.` }; const lastRow = sh.getLastRow(); const lastCol = sh.getLastColumn(); if (lastRow < 2) return { ok: false, message: "Data kosong." }; // Ambil header row 1 const headerRow = sh.getRange(1, 1, 1, lastCol).getDisplayValues()[0]; const headerMap = buildHeaderMap_(headerRow); // Ambil data (row 2 ke bawah) const values = sh.getRange(2, 1, lastRow - 1, lastCol).getDisplayValues(); // 1) Cuba cari kolum IC guna header "IC" let icCol = headerMap["IC"] || null; // Kalau jumpa kolum IC, validate sama ada nampak macam IC (12 digit) if (icCol) { const ok = sampleLooksLikeIC_(values, icCol); if (!ok) icCol = null; // header IC tapi data bukan IC -> fallback scan } // 2) Kalau tak confident, scan semua kolum cari yang match IC input if (!icCol) { icCol = detectICColumnByMatchingInput_(values, ic); if (!icCol) return { ok: false, message: "Kolum IC tidak dapat dikesan (header/data tak konsisten)." }; } // 3) Cari row yang match IC const rowIndex = findRowByIC_(values, icCol, ic); if (rowIndex === -1) return { ok: false, message: "Rekod tidak dijumpai." }; const row = values[rowIndex]; // array 0-indexed // Helper to get value by header name, fallback empty const getByHeader = (h) => { const col = headerMap[h]; if (!col) return "-"; return row[col - 1] || "-"; }; // Untuk IC: ambil dari icCol yang kita detect (paling reliable) const icValue = normalizeIC_(row[icCol - 1]) || ic; // Build output ikut order yang user nak return { ok: true, ic: icValue, nama: getByHeader("NAMA"), kelas_2025: getByHeader("KELAS_2025"), kelas_2026: getByHeader("KELAS_2026"), jantina: getByHeader("JANTINA"), guru_tingkatan_2026: getByHeader("GURU_TINGKATAN_2026") }; } function buildHeaderMap_(headers) { // map headerName -> colIndex (1-based) const map = {}; headers.forEach((h, idx) => { const key = String(h || "").trim().toUpperCase(); if (key) map[key] = idx + 1; }); // Pastikan key standard wujud walaupun header ada spaces kecil // e.g "GURU_TINGKATAN_2026" return map; } function sampleLooksLikeIC_(values, colIndex) { // check 10 row pertama: kalau banyak yang 12-digit, ok const n = Math.min(values.length, 10); let hits = 0; for (let i = 0; i < n; i++) { const v = normalizeIC_(values[i][colIndex - 1]); if (v && v.length === 12) hits++; } return hits >= Math.max(2, Math.floor(n / 3)); // cukup munasabah } function detectICColumnByMatchingInput_(values, ic) { // scan semua col, cari col yang ada exact match IC if (!values.length) return null; const colCount = values[0].length; for (let c = 1; c <= colCount; c++) { for (let r = 0; r < values.length; r++) { const cell = normalizeIC_(values[r][c - 1]); if (cell === ic) return c; } } // fallback 2: cari col yang nampak banyak 12-digit let bestCol = null; let bestScore = 0; for (let c = 1; c <= colCount; c++) { let score = 0; const n = Math.min(values.length, 30); for (let r = 0; r < n; r++) { const cell = normalizeIC_(values[r][c - 1]); if (cell && cell.length === 12) score++; } if (score > bestScore) { bestScore = score; bestCol = c; } } return bestScore >= 3 ? bestCol : null; } function findRowByIC_(values, icCol, ic) { for (let i = 0; i < values.length; i++) { const cell = normalizeIC_(values[i][icCol - 1]); if (cell === ic) return i; } return -1; } function normalizeIC_(v) { let s = String(v ?? "").trim().replace(/[^\d]/g, ""); if (s && s.length < 12) s = s.padStart(12, "0"); return s; } function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
